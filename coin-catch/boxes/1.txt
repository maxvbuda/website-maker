<body>
  <main>
    <h1>Coin <span>Catch!</span></h1>
    <p class="sub">Catch the coins. Dodge the rocks!</p>

    <div class="hud">
      <div>Score: <span class="score" id="score">0</span></div>
      <div class="lives" id="lives" aria-label="3 lives">♥ ♥ ♥</div>
      <button id="restart">Restart</button>
    </div>

    <canvas id="game" width="640" height="480" aria-label="Game area"></canvas>

    <div class="controls">
      <button class="move" data-dir="-1" aria-label="Move left">◀</button>
      <button class="move" data-dir="1" aria-label="Move right">▶</button>
    </div>

    <p class="help">Use ← → or A / D. On a phone, hold the arrows.</p>
  </main>

  <script>
    (() => {
      const canvas = document.querySelector('#game');
      const ctx = canvas.getContext('2d');
      const scoreEl = document.querySelector('#score');
      const livesEl = document.querySelector('#lives');
      const W = canvas.width;
      const H = canvas.height;

      let player, drops, score, lives, keys, over, last, spawn, raf;

      const stars = Array.from({ length: 55 }, () => ({
        x: Math.random() * W,
        y: Math.random() * H,
        r: Math.random() * 1.5 + .4,
        a: Math.random() * .65 + .2
      }));

      function reset() {
        cancelAnimationFrame(raf);
        player = { x: W / 2, y: H - 48, w: 52, h: 22, speed: 390 };
        drops = [];
        score = 0;
        lives = 3;
        keys = {};
        over = false;
        last = 0;
        spawn = 0;
        scoreEl.textContent = '0';
        livesEl.textContent = '♥ ♥ ♥';
        livesEl.setAttribute('aria-label', '3 lives');
        raf = requestAnimationFrame(loop);
      }

      function addDrop() {
        const coin = Math.random() < .68;
        const r = coin ? 13 : 16;

        drops.push({
          x: 24 + Math.random() * (W - 48),
          y: -25,
          r,
          coin,
          speed: (coin ? 145 : 180) + Math.min(score * 2, 130),
          spin: Math.random() * 6
        });
      }

      function roundedRect(x, y, w, h, r, color) {
        ctx.fillStyle = color;
        ctx.beginPath();
        ctx.roundRect(x, y, w, h, r);
        ctx.fill();
      }

      function draw() {
        const bg = ctx.createLinearGradient(0, 0, 0, H);
        bg.addColorStop(0, '#20274d');
        bg.addColorStop(1, '#17162a');
        ctx.fillStyle = bg;
        ctx.fillRect(0, 0, W, H);

        for (const s of stars) {
          ctx.globalAlpha = s.a;
          ctx.fillStyle = '#fff';
          ctx.beginPath();
          ctx.arc(s.x, s.y, s.r, 0, Math.PI * 2);
          ctx.fill();
        }
        ctx.globalAlpha = 1;

        ctx.fillStyle = '#ffffff16';
        ctx.fillRect(0, H - 16, W, 2);

        ctx.shadowColor = '#ff70b7';
        ctx.shadowBlur = 18;
        roundedRect(player.x - player.w / 2, player.y, player.w, player.h, 8, '#ff70b7');
        ctx.shadowBlur = 0;
        ctx.fillStyle = '#ffd6ea';
        ctx.fillRect(player.x - player.w / 2 + 6, player.y + 4, player.w - 12, 3);

        for (const d of drops) {
          ctx.save();
          ctx.translate(d.x, d.y);

          if (d.coin) {
            ctx.rotate(Math.sin(d.spin) * .22);
            ctx.shadowColor = '#ffd45c';
            ctx.shadowBlur = 18;
            ctx.fillStyle = '#ffd45c';
            ctx.beginPath();
            ctx.arc(0, 0, d.r, 0, Math.PI * 2);
            ctx.fill();
            ctx.shadowBlur = 0;
            ctx.fillStyle = '#ad6a13';
            ctx.font = 'bold 18px system-ui';
            ctx.textAlign = 'center';
            ctx.textBaseline = 'middle';
            ctx.fillText('$', 0, 1);
          } else {
            ctx.rotate(d.spin * .25);
            ctx.fillStyle = '#a6a9bc';
            ctx.beginPath();
            ctx.moveTo(-d.r * .9, -d.r * .55);
            ctx.lineTo(-d.r * .25, -d.r);
            ctx.lineTo(d.r * .8, -d.r * .6);
            ctx.lineTo(d.r, d.r * .2);
            ctx.lineTo(d.r * .35, d.r);
            ctx.lineTo(-d.r * .8, d.r * .65);
            ctx.closePath();
            ctx.fill();
            ctx.fillStyle = '#d9dbe5';
            ctx.beginPath();
            ctx.arc(-4, -5, 3, 0, 7);
            ctx.fill();
          }

          ctx.restore();
        }

        if (over) {
          ctx.fillStyle = '#0d1020c9';
          ctx.fillRect(0, 0, W, H);
          ctx.textAlign = 'center';
          ctx.fillStyle = '#fff';
          ctx.font = '800 42px system-ui';
          ctx.fillText('GAME OVER', W / 2, H / 2 - 16);
          ctx.fillStyle = '#ffd45c';
          ctx.font = '700 22px system-ui';
          ctx.fillText(`You caught ${score} coin${score === 1 ? '' : 's'}!`, W / 2, H / 2 + 24);
          ctx.fillStyle = '#d4d1e6';
          ctx.font = '16px system-ui';
          ctx.fillText('Press Restart to play again', W / 2, H / 2 + 58);
        }
      }

      function loop(time) {
        const dt = Math.min((time - last) / 1000 || 0, .035);
        last = time;

        if (!over) {
          if (keys.left) player.x -= player.speed * dt;
          if (keys.right) player.x += player.speed * dt;
          player.x = Math.max(player.w / 2, Math.min(W - player.w / 2, player.x));

          spawn -= dt;
          if (spawn <= 0) {
            addDrop();
            spawn = Math.max(.3, .78 - score * .008);
          }

          for (const d of drops) {
            d.y += d.speed * dt;
            d.spin += dt * 4;

            if (
              d.y + d.r >= player.y &&
              d.y - d.r <= player.y + player.h &&
              Math.abs(d.x - player.x) < player.w / 2 + d.r * .6
            ) {
              if (d.coin) score++;
              else lives--;
              d.hit = true;
            }
          }

          drops = drops.filter(d => {
            if (d.hit) return false;
            if (d.y - d.r > H) {
              if (d.coin) lives--;
              return false;
            }
            return true;
          });

          scoreEl.textContent = score;
          livesEl.textContent = '♥ '.repeat(Math.max(0, lives)).trim() || '♡';
          livesEl.setAttribute('aria-label', `${lives} lives`);
          if (lives <= 0) over = true;
        }

        draw();
        raf = requestAnimationFrame(loop);
      }

      window.addEventListener('keydown', e => {
        if (['ArrowLeft', 'ArrowRight', ' ', 'a', 'A', 'd', 'D'].includes(e.key)) {
          e.preventDefault();
        }
        if (e.key === 'ArrowLeft' || e.key.toLowerCase() === 'a') keys.left = true;
        if (e.key === 'ArrowRight' || e.key.toLowerCase() === 'd') keys.right = true;
      });

      window.addEventListener('keyup', e => {
        if (e.key === 'ArrowLeft' || e.key.toLowerCase() === 'a') keys.left = false;
        if (e.key === 'ArrowRight' || e.key.toLowerCase() === 'd') keys.right = false;
      });

      document.querySelectorAll('.move').forEach(btn => {
        const dir = Number(btn.dataset.dir);
        const down = e => {
          e.preventDefault();
          keys[dir < 0 ? 'left' : 'right'] = true;
        };
        const up = e => {
          e.preventDefault();
          keys[dir < 0 ? 'left' : 'right'] = false;
        };

        btn.addEventListener('pointerdown', down);
        ['pointerup', 'pointerleave', 'pointercancel'].forEach(t =>
          btn.addEventListener(t, up)
        );
      });

      canvas.addEventListener('pointermove', e => {
        if (e.buttons) {
          const r = canvas.getBoundingClientRect();
          player.x = (e.clientX - r.left) * W / r.width;
        }
      });

      document.querySelector('#restart').addEventListener('click', reset);
      reset();
    })();
  </script>
</body>
